cmake_minimum_required(VERSION 3.24)
project(GoldSrcFastDL VERSION 1.5.1 LANGUAGES CXX)

if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "GoldSrc requires an x86 build. Use -A Win32 or cmake/linux-x86.cmake.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

include(cmake/dependencies.cmake)
find_package(Threads REQUIRED)
add_library(fastdl_core STATIC src/config.cpp src/assets.cpp src/http.cpp src/http_response.cpp)
target_compile_features(fastdl_core PRIVATE cxx_std_20)
target_include_directories(fastdl_core PUBLIC src)
target_include_directories(fastdl_core SYSTEM PRIVATE "${boost_headers_SOURCE_DIR}")
target_link_libraries(fastdl_core PRIVATE Threads::Threads)
target_compile_definitions(fastdl_core PRIVATE BOOST_ALL_NO_LIB)
if(WIN32)
  target_compile_definitions(fastdl_core PRIVATE NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(fastdl_core PRIVATE ws2_32 mswsock)
else()
  target_compile_definitions(fastdl_core PRIVATE _FILE_OFFSET_BITS=64)
endif()

add_library(fastdl_mm MODULE src/metamod/plugin.cpp)
target_link_libraries(fastdl_mm PRIVATE fastdl_core)
target_compile_definitions(fastdl_mm PRIVATE FASTDL_VERSION="${PROJECT_VERSION}")
target_include_directories(fastdl_mm SYSTEM PRIVATE ${sdk_include_dirs})
set_target_properties(fastdl_mm PROPERTIES PREFIX "")
if(WIN32)
  target_sources(fastdl_mm PRIVATE src/metamod/exports.def)
else()
  target_compile_options(fastdl_mm PRIVATE -mstackrealign)
  target_link_options(fastdl_mm PRIVATE "LINKER:--version-script,${CMAKE_CURRENT_SOURCE_DIR}/src/metamod/exports.map" "LINKER:-z,defs")
  if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_link_options(fastdl_mm PRIVATE -static-libstdc++ -static-libgcc)
  endif()
endif()
install(TARGETS fastdl_mm LIBRARY DESTINATION addons/fastdl RUNTIME DESTINATION addons/fastdl)
install(FILES config/fastdl.ini DESTINATION addons/fastdl)
install(FILES "${boost_headers_SOURCE_DIR}/LICENSE_1_0.txt" DESTINATION addons/fastdl/licenses RENAME Boost.txt)
install(FILES "${metamod_SOURCE_DIR}/LICENSE" DESTINATION addons/fastdl/licenses RENAME Metamod-R.txt)
install(FILES README.md DESTINATION addons/fastdl)

option(BUILD_TESTING "Build the optional Metamod test host" OFF)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(fastdl_host tests/host.cpp)
  target_include_directories(fastdl_host SYSTEM PRIVATE ${sdk_include_dirs})
  target_link_libraries(fastdl_host PRIVATE ${CMAKE_DL_LIBS})
  add_test(NAME fastdl_http COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py"
    "$<TARGET_FILE:fastdl_host>" "$<TARGET_FILE:fastdl_mm>")
  set_tests_properties(fastdl_http PROPERTIES TIMEOUT 60)
endif()
